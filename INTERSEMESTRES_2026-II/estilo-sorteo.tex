		\fill[fondo] (current page.north west) rectangle (current page.south east);
		\node[anchor=north east] at ([xshift=-1cm,yshift=-0.45cm]current page.north east)
			{\includegraphics[height=1.5cm]{assets/logo-carrera.png}};
		\node[anchor=north west, align=left, text=titulo] at ([xshift=1.1cm,yshift=-0.55cm]current page.north west) {%

% Tarjeta blanca: #1 titulo, #2 subtitulo gris, #3 contenido, #4 grupo de altura
\newcommand{\tarjeta}[4]{%
	\begin{tcolorbox}[enhanced, width=\linewidth, colback=white, colframe=marco, boxrule=0.6pt, arc=2mm,
		left=3mm, right=3mm, top=2.5mm, bottom=2.5mm, equal height group=#4]
		{\blackfont\Large\color{navy} #1}{\color{muted}\small\ #2}\par\vspace{5pt}
		#3
	\end{tcolorbox}%
}
\newcommand{\grupo}[2]{\tarjeta{Grupo #1}{}{#2}{cajas}}

% Partido de llave: #1 nodo, #2 x, #3 y, #4 etiqueta, #5 local, #6 visita
\newcommand{\partido}[6]{%
	\node[font=\small\bfseries, text=acento, anchor=south west] at (#2-2,#3+0.62) {#4};
	\node[fill=white, draw=marco, rounded corners=2pt, minimum width=4cm, minimum height=0.56cm,
		font=\small\bfseries, text=ink, anchor=west] (#1h) at (#2-2,#3+0.31) {#5};
	\node[fill=white, draw=marco, rounded corners=2pt, minimum width=4cm, minimum height=0.56cm,
		font=\small\bfseries, text=ink, anchor=west] (#1a) at (#2-2,#3-0.31) {#6};
	\coordinate (#1) at (#2+2,#3);
}
\newcommand{\pasa}[2]{\draw[trazo, line width=1.1pt] (#1) -- ++(0.4,0) |- (#2.west);}
\newcommand{\ronda}[2]{\node[font=\blackfont\large, text=titulo] at #1 {#2};}

% Casillero del campeon: #1 nodo de la final, #2 x, #3 y
\newcommand{\campeon}[3]{%
	\node[fill=amarillo, rounded corners=2pt, font=\blackfont\large, text=navy, anchor=west] (camp) at (#2,#3) {Campeón};
	\draw[trazo, line width=1.4pt] (#1) -- (camp.west);
	\node[fill=white, draw=marco, rounded corners=2pt, minimum width=4.6cm, minimum height=0.7cm, anchor=north west] at (#2,#3-0.5) {};
}
