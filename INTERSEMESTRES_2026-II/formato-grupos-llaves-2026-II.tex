\newcommand{\fila}[2]{{\color{imegray}\footnotesize #1#2} & \color{imegray!50}\rule[-1pt]{0.66\linewidth}{0.4pt} & \color{imegray!50}\rule[-1pt]{0.7cm}{0.4pt}\\}

% Tabla de grupo: #1 letra, #2 numero de equipos
\newcommand{\grupo}[2]{%
	\begin{tcolorbox}[enhanced, width=\linewidth, colback=white, colframe=imegreenDark, boxrule=0.8pt, arc=2mm,
		left=1.5mm, right=1.5mm, top=1mm, bottom=1mm, fonttitle=\bfseries, coltitle=white,
		colbacktitle=imegreenDark, title={Grupo #1}, halign title=center, equal height group=grupos\thepage,
		valign lower=bottom, segmentation hidden]
		\renewcommand{\arraystretch}{1.55}
		\begin{tabular}{@{}l@{\hspace{1.5mm}}l@{\hspace{1mm}}c@{}}
			& {\footnotesize\color{imegray} Equipo} & {\footnotesize\color{imegray} Pts}\\[-3pt]
			\fila{#1}{1}\fila{#1}{2}\fila{#1}{3}\ifnum#2=4 \fila{#1}{4}\fi
		\end{tabular}
		\tcblower
		{\scriptsize\color{imegray} \ifnum#2=4 6 partidos (3 fechas)\else 3 partidos (3 fechas)\fi\ · pasan 1.° y 2.°}
	\end{tcolorbox}%
}

% Partido de llave: #1 nombre del nodo, #2 x, #3 y, #4 etiqueta, #5 local, #6 visita
\newcommand{\partido}[6]{%
	\node[font=\scriptsize\bfseries, text=imegreen, anchor=south west] at (#2-1.75,#3+0.52) {#4};
	\node[draw=imegreenDark, fill=white, rounded corners=1.5pt, minimum width=3.5cm, minimum height=0.5cm,
		font=\footnotesize, anchor=west] (#1h) at (#2-1.75,#3+0.26) {#5};
	\node[draw=imegreenDark, fill=white, rounded corners=1.5pt, minimum width=3.5cm, minimum height=0.5cm,
		font=\footnotesize, anchor=west] (#1a) at (#2-1.75,#3-0.26) {#6};
	\coordinate (#1) at (#2+1.75,#3);
}
